{{- /* Page-bundle raster images become responsive WebP; anything else (SVG, static, external) is left as-is */ -}} {{- $img := .Page.Resources.Get .Destination -}} {{- if and $img (in (slice "jpeg" "png") $img.MediaType.SubType) -}} {{- $widths := slice -}} {{- range slice 480 960 1600 -}} {{- if le . $img.Width }}{{ $widths = $widths | append . }}{{ end -}} {{- end -}} {{- if not $widths }}{{ $widths = slice $img.Width }}{{ end -}} {{- $srcset := slice -}} {{- $largest := "" -}} {{- range $widths -}} {{- $largest = $img.Resize (printf "%dx webp q78" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $largest.RelPermalink .) -}} {{- end -}} {{- else -}} {{- end -}}
